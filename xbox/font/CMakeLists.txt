cmake_minimum_required(VERSION 3.20)
project(XboxGuestFonts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature ZLIB BZIP2 PNG HARFBUZZ BROTLI)
  set(FT_DISABLE_${feature} ON CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(../../externals/freetype freetype)
if(CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
  target_compile_definitions(freetype PRIVATE _WINRT_DLL)
endif()
add_library(guest-fonts STATIC ../GuestFontLibrary.cpp)
target_link_libraries(guest-fonts PUBLIC freetype)
target_include_directories(guest-fonts PUBLIC ..)
option(XBOX_FONT_TESTS "Run portable font rendering and PS4 ABI checks" OFF)
if(XBOX_FONT_TESTS)
  enable_testing()
  add_executable(font-abi-test ../tests/FontLibraryTest.cpp)
  target_link_libraries(font-abi-test PRIVATE guest-fonts)
  add_test(NAME font-abi-render COMMAND font-abi-test "${CMAKE_CURRENT_SOURCE_DIR}/../../src/imgui/renderer/fonts")
endif()
