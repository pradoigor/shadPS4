cmake_minimum_required(VERSION 3.20)
project(XboxRuntimeChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(symbol-identity-test GuestSymbolIdentityTest.cpp)
target_include_directories(symbol-identity-test PRIVATE ..)
add_test(NAME guest-symbol-identity COMMAND symbol-identity-test)
if(WIN32)
  add_executable(sysv-abi-test SysvThunkTest.cpp ../SysvThunk.cpp)
  target_include_directories(sysv-abi-test PRIVATE .. ../../src)
  target_compile_definitions(sysv-abi-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(sysv-abi-test PRIVATE onecore)
  add_test(NAME sysv-call-and-prx-forward COMMAND sysv-abi-test)
endif()
